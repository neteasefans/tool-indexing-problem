# Here is the C compiler.
CC = gcc

# Here are the compiler options.
CFLAGS = -O2 -std=c11 -Wall -Wextra

# Here are the static linking options.
LDFLAGS = -static -static-libgcc

# Here is the .c source file.
SOURCE = SLTS_parallel5.c

# The executable file.
TARGET = SLTS5.exe

# The build and cleanup targets.
.PHONY: all clean

# make all: build the executable file.
all: $(TARGET)

# Compile and link the .c source file.
# 1 compiler 2 compiler options 3 linker options 4 -o 5 executable file 6 source file
$(TARGET): $(SOURCE)
	$(CC) $(CFLAGS) $(LDFLAGS) -o "$(TARGET)" "$(SOURCE)"

# make clean: remove the generated executable file.
clean:
ifeq ($(OS),Windows_NT)
# Use PowerShell on Windows.
	powershell -NoProfile -Command "if (Test-Path -LiteralPath '$(TARGET)') { Remove-Item -LiteralPath '$(TARGET)' -Force }"
else
# Use rm on Linux and macOS.
	rm -f "$(TARGET)"
endif
